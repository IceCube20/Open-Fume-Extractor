cmake_minimum_required(VERSION 3.20)
project(ofe_jbc_detail_ui LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG OFF CACHE BOOL "" FORCE)
set(LV_FETCH_DEPENDENCIES OFF CACHE BOOL "" FORCE)
set(LV_USE_FIND_PACKAGE OFF CACHE BOOL "" FORCE)
set(LV_USE_PKG_CONFIG OFF CACHE BOOL "" FORCE)
set(LV_BUILD_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_PATH}" lvgl-build)
get_filename_component(OFE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(DISPLAY "${OFE_ROOT}/Module/DisplayModule_800x480")
execute_process(COMMAND "${OFE_PYTHON}" "${CMAKE_CURRENT_SOURCE_DIR}/extract_settings.py" "${CMAKE_CURRENT_BINARY_DIR}/backlight_controls.inc.h" RESULT_VARIABLE EXTRACT_RESULT)
if(NOT EXTRACT_RESULT EQUAL 0)
  message(FATAL_ERROR "Backlight source extraction failed")
endif()
foreach(TARGET jbc_detail_ui jbc_detail_ui_st7796)
  add_executable(${TARGET} main.cpp "${DISPLAY}/openfume_montserrat_14_de.c" "${DISPLAY}/openfume_montserrat_18_de.c" "${DISPLAY}/openfume_montserrat_22_de.c" "${DISPLAY}/openfume_montserrat_28_de.c")
  target_include_directories(${TARGET} PRIVATE "${DISPLAY}/src" "${DISPLAY}" "${CMAKE_CURRENT_BINARY_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/../display_hybrid_native/stubs")
  target_compile_definitions(${TARGET} PRIVATE PROGMEM=)
  target_link_libraries(${TARGET} PRIVATE lvgl::lvgl)
  target_compile_options(${TARGET} PRIVATE /utf-8)
endforeach()
target_compile_definitions(jbc_detail_ui_st7796 PRIVATE ST7796_PERCEPTUAL_BACKLIGHT=1)
